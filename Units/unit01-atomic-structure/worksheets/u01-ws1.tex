\problem[1.1]{Mass of \SI{2.0}{\mole} of \ce{NaOH}
($M = \SI{40.0}{\gram\per\mole}$): \answerblank[3.5cm]{\SI{80.}{\gram}}}

\problem[1.1]{Moles in \SI{22}{\gram} of \ce{CO2}:
\answerblank[3.5cm]{\SI{0.50}{\mole}}}

\problem[1.1]{Molecules in \SI{0.50}{\mole} of \ce{H2O}:
\answerblank[3.5cm]{$3.0\times10^{23}$}}

\problem[1.1]{Total atoms in \SI{1.0}{\mole} of \ce{CaCl2} (formula units
count 3 atoms each): \answerblank[3.5cm]{$1.8\times10^{24}$}}

\problem[1.1]{A tablet contains \SI{500.0}{\milli\gram} of \ce{CaCO3}
($M = \SI{100.09}{\gram\per\mole}$).}
\begin{parts}
  \item Moles of \ce{CaCO3}: \answerblank[3.5cm]{$4.996\times10^{-3}$ mol}
  \item Moles of oxygen atoms: \answerblank[3.5cm]{$1.499\times10^{-2}$ mol}
  \item Mass of calcium in the tablet: \workspace[2.2cm]
        \keyonly{\begin{apcanswer}$4.996\times10^{-3}\,\text{mol}\times
        \SI{40.08}{\gram\per\mole}=\SI{0.2003}{\gram}$\end{apcanswer}}
\end{parts}

\problem[1.3]{Percent by mass of nitrogen in \ce{NH4NO3}
($M = \SI{80.05}{\gram\per\mole}$): \workspace[2cm]
\keyonly{\begin{apcanswer}$28.02/80.05\times100\%=35.00\%$\end{apcanswer}}}

\problem[1.3]{A compound is 43.6\% P and 56.4\% O by mass.}
\begin{parts}
  \item Empirical formula: \workspace[2.4cm]
        \keyonly{\begin{apcanswer}P: $43.6/30.97=1.41$; O: $56.4/16.00=3.53$;
        $3.53/1.41=2.5 \to \times2 \to$ \ce{P2O5}\end{apcanswer}}
  \item The molar mass is $\approx\SI{284}{\gram\per\mole}$. Molecular
        formula: \answerblank[3cm]{\ce{P4O10}}
\end{parts}
