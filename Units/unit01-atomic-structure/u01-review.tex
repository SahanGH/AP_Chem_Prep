\section*{\sffamily The five tools}

\renewcommand{\arraystretch}{1.5}
\noindent\begin{tabular}{@{}p{4.1cm}p{9.6cm}@{}}
\toprule
\textbf{Mole conversions} & $n = m/M$; \quad particles $= n \times N_A$;
  \quad always route \emph{through moles}\\
\textbf{Average atomic mass} & $\bar m = \sum(\text{abundance}\times\text{mass})$
  --- weighted, never a plain average; leans toward the abundant isotope\\
\textbf{Empirical formula} & \%\,$\to$ grams $\to$ moles $\to$ divide by
  smallest $\to$ multiply to whole numbers; molecular $= $ empirical
  $\times\, (M/M_{\text{emp}})$\\
\textbf{PES reading} & position $=$ binding energy (subshell); area $=$
  electron count; total area identifies the element\\
\textbf{Trend explanations} & three levers only: nuclear charge $Z$,
  distance/shell $n$, shielding. Rank $\to$ claim $\to$ Coulombic sentence\\
\bottomrule
\end{tabular}

\section*{\sffamily Decision tree: which formula problem is this?}

\begin{itemize}[leftmargin=*,itemsep=0.2em]
  \item Given formula, want \% $\to$ mass of element in 1 mol $\div$ molar
        mass.
  \item Given \%, want formula $\to$ the four-step recipe (assume 100 g).
  \item Given empirical + molar mass $\to$ divide, multiply subscripts.
\end{itemize}

\section*{\sffamily The traps, one line each}

\begin{itemize}[leftmargin=*,itemsep=0.25em]
  \item Unweighted isotope averages. The setup must show abundances.
  \item Mass spectrum vs PES: mass-spec height $=$ abundance of an
        \emph{isotope}; PES area $=$ electrons in a \emph{subshell}.
  \item Ion configurations: build the ion, don't label the atom.
        \ce{Ca^2+} is $[\ce{Ar}]$, not $[\ce{Ar}]4s^2$.
  \item Excited vs impossible: a vacancy below an occupied orbital is
        excited; more than $2/6/10$ in a subshell is impossible.
  \item ``Octet'' and ``wants to be stable'' are not explanations. Every
        trend answer needs $Z$, distance, or shielding.
  \item Ratios ending in .33 / .5 / .67: multiply, never round.
\end{itemize}

\section*{\sffamily Self-check --- 8 questions, exam conditions, no calculator}

\begin{enumerate}[leftmargin=*,itemsep=0.45em]
  \item Mass of \SI{0.25}{\mole} \ce{CaCO3} ($M=100$):
        \answerblank[3cm]{\SI{25}{\gram}}
  \item Atoms of H in \SI{1.0}{\mole} \ce{CH4}:
        \answerblank[3cm]{$2.4\times10^{24}$}
  \item Peaks at 24 (79\%), 25 (10\%), 26 (11\%): average?
        \answerblank[3cm]{24.3 u (Mg)}
  \item Empirical formula, 27.3\% C / 72.7\% O:
        \answerblank[3cm]{\ce{CO2}}
  \item Ground-state config of \ce{S^2-}:
        \answerblank[3.5cm]{$[\ce{Ar}]$}
  \item PES areas 2:2:6:2:1 --- element?
        \answerblank[3cm]{Al (13 e$^-$)}
  \item Larger first IE: N or O? Why, five words max.
        \answerblank[5.5cm]{N --- O's paired 2p e$^-$ repel}
  \item Formula: barium + nitrogen:
        \answerblank[3cm]{\ce{Ba3N2}}
\end{enumerate}
