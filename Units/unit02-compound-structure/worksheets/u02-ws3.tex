\problem[2.7]{\ce{CH4}, \ce{NH3}, and \ce{H2O} all have four electron
domains, yet their bond angles run 109.5$^\circ$, 107$^\circ$,
104.5$^\circ$. Write the three-sentence grader answer: count, distinguish,
conclude. \answerlines[3]{All three have 4 domains in a tetrahedral
arrangement. They differ in lone pairs: 0, 1, 2. Lone pairs repel more
strongly than bonding pairs, so each added lone pair compresses the angle
further.}}

\problem[2.7]{$\sigma$/$\pi$ bookkeeping for acetylene, \ce{HC#CH}:}
\begin{parts}
  \item Total $\sigma$: \answerblank[1.5cm]{3}\quad Total $\pi$:
        \answerblank[1.5cm]{2}
  \item Hybridization of each C: \answerblank[2cm]{sp}
  \item Every atom lies on one line. Connect that to the domain count.
        \answerlines[2]{Each C has 2 domains $\to$ 180$^\circ$ at both
        carbons $\to$ the whole molecule is linear.}
\end{parts}

\problem[2.7]{Polarity verdicts --- geometry argument required:}
\begin{parts}
  \item \ce{CCl4}: \answerblank[9.5cm]{nonpolar --- 4 equal dipoles,
        tetrahedral, cancel}
  \item \ce{CH2Cl2}: \answerblank[8.3cm]{polar --- unequal corners, no
        cancellation}
  \item \ce{XeF2} (linear, 3 LP on Xe): \answerblank[8.0cm]{nonpolar --- two
        opposite dipoles cancel}
\end{parts}

\problem[2.7]{A molecule \ce{AB3} is measured to be nonpolar with
120$^\circ$ angles; another \ce{AB3} molecule is polar with angles near
107$^\circ$. Deduce the electron-domain picture of each central atom.
\answerlines[3]{Nonpolar/120$^\circ$: 3 domains, no lone pairs (trigonal
planar, symmetric). Polar/107$^\circ$: 4 domains with 1 lone pair (trigonal
pyramidal) --- the lone pair breaks the symmetry.}}
