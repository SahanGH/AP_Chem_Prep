\section*{\sffamily The five tools}

\renewcommand{\arraystretch}{1.5}
\noindent\begin{tabular}{@{}p{4.2cm}p{9.5cm}@{}}
\toprule
\textbf{Bond classifier} & $\Delta$EN small + nonmetals $\to$ covalent
  (polar if unequal); metal + nonmetal $\to$ ionic; metal + metal $\to$
  metallic. It's a continuum --- argue with $\Delta$EN\\
\textbf{PE curve} & minimum's position $=$ bond length; depth $=$ bond
  energy; short-range wall $=$ nuclear repulsion. Breaking bonds always
  costs energy\\
\textbf{Lattice ranking} & $|E| \propto |q_1q_2|/r$: charges first, radii
  as tiebreak\\
\textbf{Lewis recipe} & count (charge-adjusted!) $\to$ skeleton (least EN
  central, never H) $\to$ outer octets $\to$ leftovers on center $\to$
  multiple bonds if short. Check count AND octets\\
\textbf{Shape chain} & Lewis $\to$ domains (multiple bond $=$ 1) $\to$
  geometry $\to$ angle (lone pairs compress) $\to$ hybridization ($2/3/4 =$
  sp/sp$^2$/sp$^3$) $\to$ polarity (dipoles + cancellation)\\
\bottomrule
\end{tabular}

\section*{\sffamily Formal charge and resonance in four lines}

\begin{itemize}[leftmargin=*,itemsep=0.2em]
  \item $FC = $ valence $-$ nonbonding $- \frac12$(bonding). Sum must equal
        the overall charge --- always check.
  \item Best structure: FCs near zero, negative FC on the most
        electronegative atom.
  \item Resonance = one static \emph{average}, never flipping. Evidence:
        equal bond lengths.
  \item Bond order $=$ bonding pairs $\div$ positions: \ce{NO3-}/\ce{CO3^2-}
        $= 4/3$; \ce{NO2-}/\ce{O3} $= 3/2$.
\end{itemize}

\section*{\sffamily The traps, one line each}

\begin{itemize}[leftmargin=*,itemsep=0.25em]
  \item Forgetting to add/subtract charge electrons in the count --- kills
        the whole structure.
  \item ``Polar bonds, so polar molecule'' --- symmetry cancels
        (\ce{CO2}, \ce{BF3}, \ce{CCl4}).
  \item Quoting 109.5$^\circ$ for \ce{NH3}/\ce{H2O} --- lone pairs compress
        (107$^\circ$/104.5$^\circ$).
  \item Ranking lattices by size when charges differ --- charges dominate.
  \item Explaining brittleness or conductivity without naming the mobile (or
        immobile) charged particles.
  \item Hybridization beyond sp$^3$ --- not assessed; don't burn time on it.
\end{itemize}

\section*{\sffamily Self-check --- 8 questions, exam conditions, no calculator}

\begin{enumerate}[leftmargin=*,itemsep=0.45em]
  \item Valence electrons in \ce{ClO3-}: \answerblank[2.9cm]{$7+18+1=26$}
  \item Most polar bond: \ce{P-H}, \ce{S-H}, \ce{O-H}:
        \answerblank[2.5cm]{\ce{O-H}}
  \item Largest lattice energy: \ce{KF}, \ce{CaO}, \ce{CaS}:
        \answerblank[2.5cm]{\ce{CaO}}
  \item Shape and angle of \ce{PH3}: \answerblank[7.3cm]{trigonal pyramidal,
        $\approx107^\circ$ ($<109.5^\circ$)}
  \item Hybridization of C in \ce{CO2}: \answerblank[2cm]{sp}
  \item $\sigma$ and $\pi$ count in \ce{N2}: \answerblank[3.5cm]{1 $\sigma$,
        2 $\pi$}
  \item Polar or nonpolar: \ce{SF2}? \answerblank[6.1cm]{polar --- bent, no
        cancellation}
  \item FC on O in \ce{H3O+}: \answerblank[2cm]{$+1$}
\end{enumerate}
